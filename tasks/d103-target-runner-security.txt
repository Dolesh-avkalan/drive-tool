def log(msg=""):
    sys.stderr.write(str(msg) + "\n")
    sys.stderr.flush()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def git_blob_sha1(data):
    h = hashlib.sha1()
    h.update(b"blob %d\0" % len(data))
    h.update(data)
    return h.hexdigest()


def sha256_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def free_bytes(path):
    st = os.statvfs(path)
    return st.f_bavail * st.f_frsize


def dir_allocated_size(path):
    total = 0
    for dp, dn, fn in os.walk(path):
        for name in fn:
            p = os.path.join(dp, name)
            try:
                st = os.lstat(p)
            except OSError:
                continue
            total += st.st_blocks * 512
    return total


def _kill_group(p):
    try:
        os.killpg(os.getpgid(p.pid), signal.SIGKILL)
    except (ProcessLookupError, PermissionError, OSError):
        try:
            p.kill()
        except OSError:
            pass


def run(cmd, cwd, env, timeout, max_output=MAX_SUBPROCESS_OUTPUT):
    """Run cmd in its own process group; capture byte-capped output; kill the
    whole group on timeout. Returns (rc, output_bytes, timed_out)."""
    p = subprocess.Popen(cmd, cwd=cwd, env=env,
                         stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                         start_new_session=True)
    buf = bytearray()
    lock = threading.Lock()

    def reader():
        while True:
            b = p.stdout.read(65536)
            if not b:
                break
            with lock:
                if len(buf) < max_output:
                    buf.extend(b[:max_output - len(buf)])

    t = threading.Thread(target=reader, daemon=True)
    t.start()
    timed_out = False
    try:
        rc = p.wait(timeout=timeout)
    except subprocess.TimeoutExpired:
        _kill_group(p)
        try:
            rc = p.wait(timeout=30)
        except subprocess.TimeoutExpired:
            rc = -9
        timed_out = True
    t.join(timeout=5)
    try:
        p.stdout.close()
    except Exception:
        pass
    with lock:
        out = bytes(buf)
    return rc, out, timed_out

def make_workspace(base_dir, workdir):
    base = os.path.realpath(base_dir)
    tmp_real = os.path.realpath("/tmp")
    # base must resolve under /tmp (or be /tmp itself)
    if not (base == tmp_real or base.startswith(tmp_real + os.sep)):
        raise SystemExit("base-dir must resolve under %s (got %s)" % (tmp_real, base))
    if workdir:
        ws = os.path.realpath(workdir)
        if not os.path.isdir(ws):
            raise SystemExit("workdir does not exist: %s" % ws)
        if not (ws.startswith(base + os.sep)):
            raise SystemExit("workdir must be a descendant of base-dir (never base itself)")
    else:
        uniq = "%d-%d" % (os.getpid(), int(time.time() * 1000) % 1000000)
        ws = os.path.join(base, "d104-%s-%s" % (COMMIT8, uniq))
        if os.path.exists(ws):
            raise SystemExit("workspace already exists: %s" % ws)
        os.makedirs(ws, mode=0o700)
        os.chmod(ws, 0o700)
    if not (ws.startswith(base + os.sep)):
        raise SystemExit("workspace escaped base-dir")
    mode = os.stat(ws).st_mode & 0o777
    if mode != 0o700:
        raise SystemExit("workspace mode is %o, expected 0700" % mode)
    return ws

def acquire_archive(archive, dest, deadline, download_timeout):
    if archive:
        src = os.path.realpath(archive)
        if not os.path.isfile(src):
            raise SystemExit("archive not found: %s" % src)
        if os.path.getsize(src) > MAX_ARCHIVE_BYTES:
            raise SystemExit("archive exceeds %d bytes" % MAX_ARCHIVE_BYTES)
        shutil.copyfile(src, dest)
        return "local:%s" % src
    remaining = max(5, min(download_timeout, int(deadline - time.monotonic())))
    cmd = ["curl", "-fsSL", "--max-time", str(remaining),
           "--retry", "2", "-o", dest, ARCHIVE_URL]
    rc, out, to = run(cmd, os.path.dirname(dest), dict(os.environ), remaining + 10)
    if rc != 0 or to:
        raise SystemExit("archive download failed rc=%s timeout=%s: %s"
                         % (rc, to, out[-400:].decode("utf-8", "replace")))
    if os.path.getsize(dest) > MAX_ARCHIVE_BYTES:
        raise SystemExit("downloaded archive exceeds %d bytes" % MAX_ARCHIVE_BYTES)
    return "download:%s" % ARCHIVE_URL


def safe_extract(tar_path, dest):
    os.makedirs(dest, mode=0o700, exist_ok=True)
    members = []
    total = 0
    with tarfile.open(tar_path, "r:gz") as tar:
        allm = tar.getmembers()
        if len(allm) > MAX_MEMBERS:
            raise SystemExit("archive has %d members > %d" % (len(allm), MAX_MEMBERS))
        for m in allm:
            name = m.name
            # reject absolute paths and traversal
            if name.startswith("/") or os.path.isabs(name):
                raise SystemExit("absolute member path: %s" % name)
            parts = name.split("/")
            if any(p == ".." for p in parts):
                raise SystemExit("path traversal member: %s" % name)
            # reject ALL links and special files
            if m.issym() or m.islnk():
                raise SystemExit("link member rejected: %s" % name)
            if not (m.isdir() or m.isfile()):
                raise SystemExit("special member rejected: %s" % name)
            if m.isfile():
                if m.size > MAX_FILE_BYTES:
                    raise SystemExit("member %s exceeds %d bytes" % (name, MAX_FILE_BYTES))
                total += m.size
                if total > MAX_TOTAL_UNCOMPRESSED:
                    raise SystemExit("uncompressed total exceeds %d bytes" % MAX_TOTAL_UNCOMPRESSED)
            members.append(m)
        for m in members:
            target = os.path.join(dest, m.name)
            real = os.path.realpath(target)
            if not (real == os.path.realpath(dest) or real.startswith(os.path.realpath(dest) + os.sep)):
                raise SystemExit("member escapes extract dir: %s" % m.name)
            if m.isdir():
                os.makedirs(target, mode=0o700, exist_ok=True)
            else:
                os.makedirs(os.path.dirname(target), mode=0o700, exist_ok=True)
                f = tar.extractfile(m)
                data = f.read() if f else b""
                with open(target, "wb") as out:
                    out.write(data)
                os.chmod(target, 0o600)
    return total

def verify_identities(src, expected):
    """Recompute + verify every executable source/test/importable file."""
    problems = []
    seen = set()
    expected_paths = set(expected.keys())
    for dp, dn, fn in os.walk(src):
        for name in fn:
            p = os.path.join(dp, name)
            rel = os.path.relpath(p, src)
            try:
                st = os.lstat(p)
            except OSError as e:
                problems.append("cannot stat %s: %s" % (rel, e))
                continue
            import stat as _stat
            if _stat.S_ISLNK(st.st_mode):
                problems.append("symlink present in source: %s" % rel)
                continue
            if not _stat.S_ISREG(st.st_mode):
                problems.append("non-regular file in source: %s" % rel)
                continue
            base = os.path.basename(rel)
            importable = base.endswith(EXECUTABLE_EXTS) or base in EXECUTABLE_NAMES
            under_dt = rel.startswith("drive_tool" + os.sep) or rel == "drive_tool"
            under_ts = rel.startswith("tests" + os.sep) or rel == "tests"
            if rel in expected_paths:
                seen.add(rel)
                data = open(p, "rb").read()
                blob = git_blob_sha1(data)
                sha = sha256_bytes(data)
                eblob, esha, esize = expected[rel]
                if len(data) != esize:
                    problems.append("size mismatch %s: %d != %d" % (rel, len(data), esize))
                if blob != eblob:
                    problems.append("blob mismatch %s: %s != %s" % (rel, blob, eblob))
                if sha != esha:
                    problems.append("sha256 mismatch %s: %s != %s" % (rel, sha, esha))
                continue
            if under_dt or under_ts:
                problems.append("unexpected file under drive_tool/ or tests/: %s" % rel)
                continue
            if importable:
                problems.append("unexpected importable/executable file: %s" % rel)
                continue
    for rel in expected_paths:
        if rel not in seen:
            problems.append("expected file missing: %s" % rel)
    return problems


def verify_gate_payload(path):
    if not os.path.isfile(path):
        return ["gate payload not found: %s" % path]
    got = sha256_file(path)
    if got != GATE_SHA256:
        return ["gate payload sha256 mismatch: %s != %s" % (got, GATE_SHA256)]
    return []

def parse_suite(text):
    ran = None
    skipped_ids = []
    failed = False
    for line in text.splitlines():
        ls = line.strip()
        if ls.startswith("Ran ") and " test" in ls:
            try:
                ran = int(ls.split()[1])
            except (ValueError, IndexError):
                pass
        if " ... skipped " in line:
            head = line.split(" ... skipped ", 1)[0].strip()
            if "(" in head and head.endswith(")"):
                tid = head[head.rindex("(") + 1:-1].strip()
            else:
                tid = head.split(" ", 1)[0]
            skipped_ids.append(tid)
        if ls.startswith("FAILED") or ls == "FAILED" or "FAILED (" in ls:
            failed = True
        if ls.startswith("ERROR") and "=" in ls:
            failed = True
    return ran, skipped_ids, failed


def suite_ok(rc, out, timed_out):
    text = out.decode("utf-8", "replace")
    ran, skips, failed = parse_suite(text)
    problems = []
    if timed_out:
        problems.append("suite timed out")

def frozen_schedule(total=SMOKE_TOTAL, valid=SMOKE_VALID, ro_save=SMOKE_RO_SAVE,
                    seed=FROZEN_SEED):

# Drain-aware lease limiter
# --------------------------------------------------------------------------
class LeaseLimiter:
    """K global client leases; each is held until response + a monotonic hold."""

    def __init__(self, capacity, hold_seconds):
        self._sem = threading.BoundedSemaphore(capacity)
        self._hold = hold_seconds
        self.max_held = 0
        self._lock = threading.Lock()
        self._held = 0

    def acquire(self):
        self._sem.acquire()
        with self._lock:
            self._held += 1
            if self._held > self.max_held:
                self.max_held = self._held

    def release(self, response_ts):
        wait = (response_ts + self._hold) - time.monotonic()
        if wait > 0:
            time.sleep(wait)
        with self._lock:
            self._held -= 1
        self._sem.release()

                    status = 200
                except DriveClientError as exc:
                    code = exc.code
                    unc = exc.outcome_uncertain
                    retry = exc.retry_safe
                except Exception as exc:      # noqa: BLE001
                    transport_exc = type(exc).__name__
                response_ts = time.monotonic()
                bucket = classify(intent.expect, status, code, transport_exc)
                local.append({
                    "seq": intent.seq, "agent": token_label(intent.agent),
                    "op": intent.op, "expect": intent.expect, "bucket": bucket,
                    "status": status, "code": code,
                    "outcome_uncertain": unc, "retry_safe": retry,
                    "transport": transport_exc,
                    "duration_ms": round((response_ts - t0) * 1000.0, 3),
                })
                limiter.release(response_ts)
            with lock:
                for rec in local:
                    buckets[rec["bucket"]] += 1
                records.extend(local)

        shards = [intents[i::leases] for i in range(leases)]
        threads = [threading.Thread(target=worker, args=(s,), name="d096-client-%d" % i)
                   for i, s in enumerate(shards)]
        for t in threads:
            t.start()
        for t in threads:
            t.join(timeout=budget)

        elapsed = time.monotonic() - started
        fd1, rss1 = _nfd(), _rss_kb()
        threads_after = _adapter_threads()
        workspace_bytes = _tree_bytes(base)

        expected = {"valid_below_capacity_success": valid,
                    "expected_readonly_denial": ro_save,
                    "expected_saturation_busy": 0,
                    "unexpected_unavailable": 0,
                    "unexpected_other_error": 0}
        ok, problems = verdict(buckets, expected)

        resources = {
            "workspace_bytes": workspace_bytes,
            "workspace_limit_bytes": MAX_WORKSPACE_BYTES,
            "fixture_bytes": fixture_bytes,
            "fixture_limit_bytes": MAX_FIXTURE_BYTES,
            "fd_before": fd0, "fd_after": fd1,
            "fd_ok": (fd0 is None or fd1 is None or fd1 <= fd0 + FD_BASELINE_SLACK),
            "adapter_threads_live": threads_live,

            "closed": closed,
            "max_leases_held": limiter.max_held,
            "elapsed_seconds": round(elapsed, 3),

        if not resources["closed"]:
            ok, problems = False, problems + ["adapter.close() did not return True"]
        for key in ("fd_ok", "adapter_threads_ok", "rss_ok", "within_budget"):
            if not resources[key]:
                ok, problems = False, problems + ["resource bound failed: " + key]
        if resources["workspace_bytes"] > MAX_WORKSPACE_BYTES:
            ok, problems = False, problems + ["workspace exceeds the frozen bound"]
        if resources["max_leases_held"] > leases:
            ok, problems = False, problems + ["client leases exceeded K"]

        records.sort(key=lambda r: r["seq"])

    finally:

def _raw_request(path, token, body, ctype=b"application/json"):
    head = (b"POST " + path.encode("ascii") + b" HTTP/1.1\r\n"
            b"Host: 127.0.0.1\r\n"
            b"Authorization: Bearer " + token.encode("ascii") + b"\r\n"
            b"Content-Type: " + ctype + b"\r\n"
            b"Content-Length: " + str(len(body)).encode("ascii") + b"\r\n"
            b"Connection: close\r\n\r\n")
    return head + body


def _raw_exchange(port, payload, *, timeout=5.0, read_delay=0.0):
    s = socket.create_connection(("127.0.0.1", port), timeout=timeout)
    try:
        s.sendall(payload)
        if read_delay:
            time.sleep(read_delay)
        s.settimeout(timeout)
        chunks = []
        while True:
            try:
                chunk = s.recv(4096)
            except socket.timeout:
                break
            if not chunk:
                break
            chunks.append(chunk)
        return b"".join(chunks)
    finally:
        s.close()


def _status_code(raw):
    if not raw:
        return None, None
    try:
        status = int(raw.split(b" ", 2)[1])
    except (IndexError, ValueError):
        return None, None
    code = None
    if b"\r\n\r\n" in raw:
        try:
            obj = json.loads(raw.split(b"\r\n\r\n", 1)[1].decode("utf-8"))
            if isinstance(obj, dict) and obj.get("ok") is False \
                    and isinstance(obj.get("error"), dict):
                code = obj["error"].get("code")
        except Exception:  # noqa: BLE001
            pass
    return status, code


# --------------------------------------------------------------------------
# Track B: deterministic 6+1 saturation (D097 correction)
#
# D096/D095 placed the excess probe inside a 50 ms drain window using a
# threading.Barrier plus a read_delay and a 10 ms excess sleep.  That is a
# timing race: if the excess connection is accepted before all six occupiers
# hold a worker slot, the excess is admitted and an occupier is rejected busy
# (the D095 "299 valid / 49 busy / 2 other" flake).  It is not deterministic.
#
# The D097 correction holds the six admitted operations at a REAL service-entry
# barrier: the harness service blocks inside its list() handler until all six
# workers have confirmed entry, the excess probe is sent only then (while all
# six slots are provably held in-handler), and the six are released only after
# the probe has been fully answered.  No socket-open / drain-window timing is
# used anywhere on the saturation path.
# --------------------------------------------------------------------------
class _EntryGate:
    """A per-cycle real service-entry barrier for exactly ``n`` workers."""

    def __init__(self, n):
        self._n = n
        self._lock = threading.Lock()
        self._count = 0
        self.entered = threading.Event()      # set once all n have entered
        self.release = threading.Event()      # main sets to let them proceed

    @property
    def count(self):
        with self._lock:
            return self._count

    def enter(self):
        with self._lock:
            self._count += 1
            if self._count >= self._n:
                self.entered.set()
        self.release.wait(timeout=30.0)


def _make_gated_service(policies):
    """LocalFileService subclass whose list() waits on the active entry gate."""
    from drive_tool.file_service import LocalFileService

    class _GatedService(LocalFileService):
        _gate = None

        def list(self, token, source_id, file_id="root", **kwargs):
            gate = self._gate
            if gate is not None:
                gate.enter()
            return super().list(token, source_id, file_id, **kwargs)

    return _GatedService(policies)


def run_saturation(repo, cycles=SAT_CYCLES, budget=STAGE_BUDGET_SECONDS):
    """Deterministic overload with a REAL service-entry barrier.

    Six admitted list operations are held inside the service handler until all
    six workers have confirmed entry; only then is the excess probe sent.  With
    every worker slot provably occupied in-handler, the excess must receive a
    503 busy envelope, and no occupier can be rejected.
    """
    sys.path.insert(0, repo)
    from drive_tool.http_adapter import DriveHTTPAdapter

    base = tempfile.mkdtemp(prefix="d097-sat-")
    started = time.monotonic()
    try:
        adapter = None
        roots, _fixture = build_roots(base)
        policies = build_policies(roots)
        service = _make_gated_service(policies)
        adapter = DriveHTTPAdapter(service=service, port=0, socket_timeout=5.0)
        time.sleep(0.1)
        fd0 = _nfd()
        threads_live = _adapter_threads()
        token = _TOKENS[0]
        payload = json.dumps({"source_id": "main", "file_id": "root",
                              "limit": 20, "offset": 0}).encode("ascii")
        req = _raw_request("/v1/list", token, payload)

        results = []
        for cycle in range(cycles):
            gate = _EntryGate(SAT_OCCUPIERS)
            service._gate = gate
            cycle_out = {}

            def occupier(slot):
                raw = _raw_exchange(adapter.port, req, timeout=15.0)
                status, code = _status_code(raw)
                cycle_out[("occupier", slot)] = (status, code)

            threads = [threading.Thread(target=occupier, args=(i,))
                       for i in range(SAT_OCCUPIERS)]
            for t in threads:
                t.start()

            entered_ok = gate.entered.wait(timeout=20.0)
            entered_count = gate.count

            # All six worker slots are provably held inside the handler here.
            exc_raw = _raw_exchange(adapter.port, req, timeout=15.0)
            exc = _status_code(exc_raw)
            cycle_out[("excess", 0)] = exc

            # Release the six only after the excess probe is complete.
            gate.release.set()
            for t in threads:
                t.join(timeout=20.0)

            service._gate = None

            occ = [cycle_out.get(("occupier", i), (None, None))
                   for i in range(SAT_OCCUPIERS)]
            results.append({"cycle": cycle, "entered_count": entered_count,
                            "entered_ok": entered_ok,
                            "occupiers": occ, "excess": exc})

        closed = adapter.close(timeout=10.0)
        elapsed = time.monotonic() - started
        fd1 = _nfd()
        threads_after = _adapter_threads()

        occ_ok = sum(1 for r in results for (s, c) in r["occupiers"] if s == 200)
        occ_bad = sum(1 for r in results for (s, c) in r["occupiers"]
                      if not (s == 200))
        exc_busy = sum(1 for r in results if r["excess"][1] == "busy")
        exc_unavail = sum(1 for r in results if r["excess"][1] == "unavailable")
        exc_other = len(results) - exc_busy - exc_unavail
        entered_misses = sum(1 for r in results if not r["entered_ok"])

        buckets = {
            "valid_below_capacity_success": occ_ok,
            "expected_readonly_denial": 0,
            "expected_saturation_busy": exc_busy,
            "unexpected_unavailable": exc_unavail,
            "unexpected_other_error": occ_bad + exc_other,
        }
        expected = {"valid_below_capacity_success": SAT_OCCUPIERS * cycles,
                    "expected_readonly_denial": 0,
                    "expected_saturation_busy": SAT_EXCESS * cycles,
                    "unexpected_unavailable": 0,
                    "unexpected_other_error": 0}
        ok, problems = verdict(buckets, expected)
        if entered_misses:
            ok, problems = False, problems + [
                "service-entry barrier not reached in %d cycle(s)" % entered_misses]
        if not closed:
            ok, problems = False, problems + ["adapter.close() did not return True"]
        if threads_after:
            ok, problems = False, problems + ["adapter threads not reaped"]
        if fd0 is not None and fd1 is not None and fd1 > fd0 + FD_BASELINE_SLACK:
            ok, problems = False, problems + ["fd did not return to baseline"]

        return {
            "harness": "d097-http-gate-harness",
            "stage": "saturation_6plus1",
            "saturation_mode": "real-service-entry-barrier",
            "frozen": {"cycles": cycles, "occupiers": SAT_OCCUPIERS,
                       "excess": SAT_EXCESS, "budget_seconds": budget},
            "requested": cycles * (SAT_OCCUPIERS + SAT_EXCESS),
            "buckets": buckets,
            "expected": expected,
            "result": "PASS" if ok else "FAIL",
            "problems": problems,
            "resources": {"fd_before": fd0, "fd_after": fd1, "closed": closed,
                          "adapter_threads_live": threads_live,
                          "adapter_threads_after_close": threads_after,
                          "elapsed_seconds": round(elapsed, 3),
                          "within_budget": elapsed <= budget},
            "cycles": results,
        }
    finally:
        if adapter is not None:
            try:
                adapter.close(timeout=5.0)
            except BaseException:
                pass
        shutil.rmtree(base, ignore_errors=True)

# --------------------------------------------------------------------------
# Gating helper: selfcheck, then smoke, then soak + saturation if smoke passes
#
# D101 correction -- HARNESS-ONLY AGGREGATE EXIT/STATUS.
#
# D096/D097 ``--gate`` returned ``{"smoke": ..., "soak": ...}`` with no
# top-level ``result`` key, so the exit mapping in ``main()`` read ``None``
# and returned a NONZERO code even when every stage PASSed (the quirk the
# D097 report records in section 8).  Nothing about the stages was wrong;
# only the aggregate exit/status was.
#
# The corrected gate now:
#   * orchestrates the four REQUIRED stages -- selfcheck, smoke600,
#     soak10000 and saturation_6plus1;
#   * returns an aggregate dict that CARRIES a top-level ``result`` key;
#   * exits 0 iff EVERY required stage was executed AND its own result is
#     PASS; any FAIL, missing, skipped, error or timeout stage makes the
#     gate nonzero;
#   * preserves the D096/D097 rule: NO soak (nor saturation) after a smoke
#     failure -- both are recorded as skipped and the gate is nonzero.
#
# Only this helper and the exit mapping that reads its ``result`` changed.
# The stage runners, the frozen schedules/intents, the D089 accounting
# buckets, the budgets and the real service-entry saturation barrier are
# byte-for-byte the D097 code.
# --------------------------------------------------------------------------
REQUIRED_STAGES = ("selfcheck", "smoke600", "soak10000", "saturation_6plus1")


def _run_guarded(fn, *args, **kwargs):
    """Run a stage, turning any exception/timeout into an explicit non-PASS.

    A stage that raises must never be able to look like a pass, so a
    TimeoutError becomes ``TIMEOUT`` and any other exception becomes
    ``ERROR``; a non-dict return becomes ``ERROR``.  All three are non-PASS
    and therefore fail the aggregate.
    """
    try:
        res = fn(*args, **kwargs)
    except TimeoutError:
        return {"result": "TIMEOUT", "problems": ["stage timed out"]}
    except BaseException as exc:      # noqa: BLE001
        return {"result": "ERROR",
                "problems": ["stage raised %s" % type(exc).__name__]}
    if not isinstance(res, dict):
        return {"result": "ERROR", "problems": ["stage returned non-dict"]}
    return res


def _aggregate_gate(stages):
    """PASS iff every REQUIRED stage was executed and reports PASS.

    A missing/skipped stage (``None``), or any result other than ``PASS``
    (FAIL / ERROR / TIMEOUT / MISSING / anything), is a problem, so a false
    pass is impossible by construction.
    """
    problems = []
    for name in REQUIRED_STAGES:
        res = stages.get(name)
        if res is None:
            problems.append("%s: not executed (skipped/missing)" % name)
            continue
        status = res.get("result")
        if status != "PASS":
            problems.append("%s: %s" % (name, "MISSING" if status is None else status))
    return (not problems), problems


def run_gate(repo, allow_diagnostic_soak=False):
    """Run the four required stages and return an aggregate verdict.

    ``allow_diagnostic_soak`` is accepted for CLI compatibility but is
    intentionally inert: D101 forbids a soak after a smoke failure, so the
    soak is always withheld when smoke does not pass.
    """
    stages = {name: None for name in REQUIRED_STAGES}

    stages["selfcheck"] = _run_guarded(selfcheck)
    if stages["selfcheck"].get("result") == "PASS":
        stages["smoke600"] = _run_guarded(
            run_stage, repo, SMOKE_TOTAL, SMOKE_VALID, SMOKE_RO_SAVE, "smoke600")

    smoke = stages["smoke600"]
    if smoke is not None and smoke.get("result") == "PASS":
        stages["soak10000"] = _run_guarded(
            run_stage, repo, SOAK_TOTAL, SOAK_VALID, SOAK_RO_SAVE, "soak10000")
        stages["saturation_6plus1"] = _run_guarded(run_saturation, repo)
    # else: smoke skipped or failed -> soak and saturation are withheld.

    ok, problems = _aggregate_gate(stages)
    note = None
    if stages["soak10000"] is None:
        note = "soak withheld: smoke did not pass (no soak after smoke fail)"

    return {
        "harness": "d101-http-gate-harness",
        "stage": "gate",
        "required_stages": list(REQUIRED_STAGES),
        "executed": [n for n in REQUIRED_STAGES if stages[n] is not None],
        "result": "PASS" if ok else "FAIL",
        "problems": problems,
        "note": note,
        "selfcheck": stages["selfcheck"],
        "smoke": stages["smoke600"],
        "soak": stages["soak10000"],
        "saturation": stages["saturation_6plus1"],
    }


# --------------------------------------------------------------------------
# Self-check (schedule + accounting), no adapter required
# --------------------------------------------------------------------------
def selfcheck():
    problems = []
    smoke = frozen_schedule(SMOKE_TOTAL, SMOKE_VALID, SMOKE_RO_SAVE)
    soak = frozen_schedule(SOAK_TOTAL, SOAK_VALID, SOAK_RO_SAVE)
    for name, intents, total, valid, ro in (
            ("smoke600", smoke, SMOKE_TOTAL, SMOKE_VALID, SMOKE_RO_SAVE),
            ("soak10000", soak, SOAK_TOTAL, SOAK_VALID, SOAK_RO_SAVE)):
        counts, per_agent = schedule_counts(intents)
        if len(intents) != total:
            problems.append("%s: %d intents != %d" % (name, len(intents), total))
        if counts["ok"] != valid:
            problems.append("%s: ok %d != %d" % (name, counts["ok"], valid))
        if counts["denied"] != ro:
            problems.append("%s: denied %d != %d" % (name, counts["denied"], ro))
        if len(per_agent) != AGENT_COUNT:
            problems.append("%s: %d personas != %d" % (name, len(per_agent), AGENT_COUNT))
        for it in intents:
            if it.agent == READ_ONLY_AGENT and it.op == "save" and it.expect != "denied":
                problems.append("%s: RO save not marked denied" % name)
            if it.expect == "ok" and it.op == "save" and it.agent == READ_ONLY_AGENT:
                problems.append("%s: RO agent given an authorized save" % name)
        again = frozen_schedule(total, valid, ro)
        if schedule_fingerprint(again) != schedule_fingerprint(intents):
            problems.append("%s: schedule is not replayable" % name)

    # accounting truth table
    table = [
        ("ok", 200, None, None, "valid_below_capacity_success"),
        ("ok", None, "busy", None, "unexpected_other_error"),
        ("ok", None, "denied", None, "unexpected_other_error"),
        ("ok", None, "unavailable", None, "unexpected_unavailable"),
        ("ok", None, "timeout", None, "unexpected_other_error"),
        ("denied", None, "denied", None, "expected_readonly_denial"),
        ("denied", 200, None, None, "unexpected_other_error"),
        ("denied", None, "busy", None, "unexpected_other_error"),
        ("denied", None, "unavailable", None, "unexpected_unavailable"),
        ("busy", None, "busy", None, "expected_saturation_busy"),
        ("busy", None, "unavailable", None, "unexpected_unavailable"),
        ("busy", 200, None, None, "unexpected_other_error"),
        ("ok", None, "unavailable", "RemoteDisconnected", "unexpected_unavailable"),
        ("ok", None, None, "OSError", "unexpected_other_error"),
    ]
    for expect, status, code, exc, want in table:
        got = classify(expect, status, code, exc)
        if got != want:
            problems.append("classify(%r,%r,%r,%r) -> %s != %s"
                            % (expect, status, code, exc, got, want))

    # saturation arithmetic
    if SAT_OCCUPIERS * SAT_CYCLES != 300:
        problems.append("saturation success arithmetic != 300")
    if SAT_EXCESS * SAT_CYCLES != 50:
        problems.append("saturation busy arithmetic != 50")
    if DRAIN_HOLD_SECONDS < 0.055:
        problems.append("drain hold below the frozen 55 ms")

    return {
        "harness": "d096-http-gate-harness",
        "stage": "selfcheck",
        "result": "PASS" if not problems else "FAIL",
        "problems": problems,
        "schedule_fingerprints": {
            "smoke600": schedule_fingerprint(smoke),
            "soak10000": schedule_fingerprint(soak),
        },
        "schedule_counts": {
            "smoke600": schedule_counts(smoke)[0],
            "soak10000": schedule_counts(soak)[0],
        },
        "saturation_arithmetic": {"valid": SAT_OCCUPIERS * SAT_CYCLES,
                                  "busy": SAT_EXCESS * SAT_CYCLES},
        "frozen": {"seed": FROZEN_SEED, "leases": CLIENT_LEASES,
                   "hold_seconds": DRAIN_HOLD_SECONDS},
    }
