    # ----- shutdown -------------------------------------------------------

    def close(self, *, timeout=DEFAULT_DRAIN_TIMEOUT_SECONDS):
        if type(timeout) is bool or not isinstance(timeout, (int, float)):
            raise ValueError("timeout must be a finite non-negative number")
        timeout_f = float(timeout)
        if not math.isfinite(timeout_f) or timeout_f < 0.0:
            raise ValueError("timeout must be a finite non-negative number")

        with self._close_lock:
            if self._closed:
                return True

            deadline = time.monotonic() + timeout_f

            with self._pool_cond:
                should_close_listener = not self._listener_closed
                self._listener_closed = True
                self._draining = True
                for slot in self._slots:
                    if slot.conn is not None:
                        try:
                            slot.conn.close()
                        except OSError:
                            pass
                        slot.conn = None
                self._pool_cond.notify_all()

            if should_close_listener:
                self._wake_listener()
                if self._listener is not None:
                    try:
                        self._listener.close()
                    except OSError:
                        pass

            remaining = max(0.0, deadline - time.monotonic())
            try:
                service_closed = self._service.close(timeout=remaining)
            except BaseException:
                service_closed = False

            if self._accept_thread is not None:
                remaining = max(0.0, deadline - time.monotonic())
                try:
                    self._accept_thread.join(timeout=remaining)
                except BaseException:
                    pass

            for worker in self._workers:
                remaining = max(0.0, deadline - time.monotonic())
                try:
                    worker.join(timeout=remaining)
                except BaseException:
                    pass

            accept_done = self._accept_thread is None or not self._accept_thread.is_alive()
            workers_done = all(not worker.is_alive() for worker in self._workers)

            if service_closed and accept_done and workers_done:
                self._closed = True
                return True
            return False

    def _wake_listener(self):
        port = self._port
        if port <= 0:
            return
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            try:
                sock.setblocking(False)
                sock.connect_ex((HOST, port))
            finally:
                sock.close()
        except Exception:
            pass

    def _accept_loop(self):
        while True:
            with self._pool_lock:
                if self._draining:
                    return
            try:
                conn, _ = self._listener.accept()
            except socket.timeout:
                continue
            except OSError:
                with self._pool_lock:
                    if self._draining:
                        return
                # transient; loop with listener timeout
                continue

            assigned = False
            with self._pool_cond:
                if self._draining:
                    try:
                        conn.close()
                    except OSError:
                        pass
                    continue
                if self._idle_slots:
                    slot = self._idle_slots.pop()
                    slot.conn = conn
                    self._pool_cond.notify_all()
                    assigned = True
            if not assigned:
                self._busy_reject(conn)
