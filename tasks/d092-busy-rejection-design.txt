      def _send_with_deadline(self, conn, data, deadline):
          view = memoryview(data)
          while len(view) > 0:
              remaining = deadline - time.monotonic()
              if remaining <= 0:
                  return False
              try:
                  conn.settimeout(remaining)
              except (OSError, ValueError):
                  return False
