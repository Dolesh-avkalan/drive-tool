def git_blob_sha1(data: bytes) -> str:
    h = hashlib.sha1()
    h.update(b"blob %d\x00" % len(data))
    h.update(data)
    return h.hexdigest()


def sha256_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as fh:
        for chunk in iter(lambda: fh.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()


def verify_identities(src, expected):
    problems = []
    for rel, want in sorted(expected.items()):
        p = os.path.join(src, rel)
        if not os.path.isfile(p):
            problems.append(f"MISSING {rel}")
            continue
        with open(p, "rb") as fh:
            got = git_blob_sha1(fh.read())
        if got != want:
            problems.append(f"MISMATCH {rel}: got {got} want {want}")
    extra = []
    for dp, dns, fns in os.walk(src):
        dns[:] = [d for d in dns if d != "__pycache__"]
        for fn in fns:
            rel = os.path.relpath(os.path.join(dp, fn), src)
            if rel not in expected:
                extra.append(rel)
    return problems, extra

def free_bytes(path):
    try:
        return shutil.disk_usage(path).free
    except OSError:
        return None


def meminfo():
    info = {}
    try:
        with open("/proc/meminfo") as fh:
            for line in fh:
                k, _, v = line.partition(":")
                info[k.strip()] = v.strip()
    except OSError:
        pass
    return info


def os_release():
    rel = {}
    try:
        with open("/etc/os-release") as fh:
            for line in fh:
                if "=" in line:
                    k, _, v = line.partition("=")
                    rel[k.strip()] = v.strip().strip('"')
    except OSError:
        pass
    return rel


def dir_size(path):
    total = 0
    for dp, dns, fns in os.walk(path):
        for fn in fns:
            p = os.path.join(dp, fn)
            try:
                total += os.path.getsize(p)
            except OSError:
                pass
    return total

def download_archive(url, dest, max_bytes, timeout):
    req = urllib.request.Request(url, headers={"User-Agent": "d063-stress-runner"})
    total = 0
    with urllib.request.urlopen(req, timeout=timeout) as resp, open(dest, "wb") as out:
        while True:
            chunk = resp.read(65536)
            if not chunk:
                break
            total += len(chunk)
            if total > max_bytes:
                raise RuntimeError(f"archive exceeds cap {max_bytes} bytes")
            out.write(chunk)
    return total


def copy_archive(src, dest, max_bytes):
    size = os.path.getsize(src)
    if size > max_bytes:
        raise RuntimeError(f"archive {size} bytes exceeds cap {max_bytes}")
    shutil.copyfile(src, dest)
    return size


def safe_extract(tar_path, dest):
    """Extract a tar.gz with path/size/count/link guards. Returns member count."""
    dest_real = os.path.realpath(dest)
    count = 0
    total = 0
    with tarfile.open(tar_path, "r:gz") as t:
        for m in t:
            count += 1
            if count > MAX_MEMBERS:
                raise RuntimeError(f"archive has more than {MAX_MEMBERS} members")
            name = m.name
            if name.startswith("/") or os.path.isabs(name):
                raise RuntimeError(f"absolute path in archive: {name!r}")
            parts = [p for p in name.split("/") if p not in ("", ".")]
            if ".." in parts or not parts:
                raise RuntimeError(f"unsafe path in archive: {name!r}")
            if m.issym() or m.islnk():
                raise RuntimeError(f"link member in archive: {name!r}")
            outpath = os.path.join(dest, *parts)
            if not os.path.realpath(outpath).startswith(dest_real + os.sep):
                raise RuntimeError(f"path escapes destination: {name!r}")
            if m.isdir():
                os.makedirs(outpath, exist_ok=True)
                continue
            if not m.isfile():
                raise RuntimeError(f"non-regular member in archive: {name!r}")
            if m.size > MAX_FILE_BYTES:
                raise RuntimeError(f"file {name!r} exceeds {MAX_FILE_BYTES} bytes")
            total += m.size
            if total > MAX_TOTAL_UNCOMPRESSED:
                raise RuntimeError("archive uncompressed size exceeds cap")
            os.makedirs(os.path.dirname(outpath), exist_ok=True)
            srcf = t.extractfile(m)
            if srcf is None:
                raise RuntimeError(f"cannot read member {name!r}")
            with srcf, open(outpath, "wb") as outf:
                shutil.copyfileobj(srcf, outf, length=65536)
    return count


def find_src_root(extract_dir):
    entries = list(os.listdir(extract_dir))
    if len(entries) != 1 or not os.path.isdir(os.path.join(extract_dir, entries[0])):
        raise RuntimeError(f"archive did not extract to a single top directory: {entries}")
    return os.path.join(extract_dir, entries[0])

def discover_pythons(requested):
    found = []
    seen = set()
    for name in (requested or PYTHON_CANDIDATES):
        path = shutil.which(name)
        if not path:
            continue
        real = os.path.realpath(path)
        if real in seen:
            continue
        seen.add(real)
        try:
            out = subprocess.run(
                [path, "-c", "import sys;print('%d.%d.%d'%sys.version_info[:3])"],
                capture_output=True, text=True, timeout=30)
            ver = out.stdout.strip() if out.returncode == 0 else "unknown"
        except Exception:  # noqa: BLE001
            ver = "unknown"
        found.append({"name": name, "path": path, "version": ver})
    return found


def run(cmd, cwd, env, timeout):
    start = time.time()
    try:
        p = subprocess.run(cmd, cwd=cwd, env=env, capture_output=True,
                           text=True, timeout=timeout)
        return {"cmd": cmd, "rc": p.returncode, "out": p.stdout, "err": p.stderr,
                "timed_out": False, "seconds": round(time.time() - start, 2)}
    except subprocess.TimeoutExpired as exc:
        def _txt(v):
            return v.decode("utf-8", "replace") if isinstance(v, bytes) else (v or "")
        return {"cmd": cmd, "rc": None, "out": _txt(exc.stdout), "err": _txt(exc.stderr),
                "timed_out": True, "seconds": round(time.time() - start, 2)}
    except Exception as exc:  # noqa: BLE001
        return {"cmd": cmd, "rc": None, "out": "", "err": f"{type(exc).__name__}: {exc}",
                "timed_out": False, "seconds": round(time.time() - start, 2)}


def parse_unittest(text):
    ran = skipped = None
    m = re.search(r"Ran (\d+) tests?", text)
    if m:
        ran = int(m.group(1))
    m = re.search(r"skipped=(\d+)", text)
    if m:
        skipped = int(m.group(1))
    elif ran is not None:
        skipped = 0
    ok = bool(re.search(r"^OK", text, re.M))
    failed = bool(re.search(r"^FAILED", text, re.M))
    return {"ran": ran, "skipped": skipped, "ok": ok, "failed": failed}


def build_env(workspace, src):
    env = dict(os.environ)
    tmp = os.path.join(workspace, "tmp")
    os.makedirs(tmp, exist_ok=True)

    env["PYTHONPATH"] = src + (os.pathsep + env["PYTHONPATH"] if env.get("PYTHONPATH") else "")
    env["D063_WORKSPACE"] = workspace
    return env

def write_manifest(workspace):
    rows = []
    for dp, dns, fns in os.walk(workspace):
        dns[:] = [d for d in dns if d != "__pycache__"]
        for fn in sorted(fns):
            p = os.path.join(dp, fn)
            rows.append((os.path.relpath(p, workspace), os.path.getsize(p), sha256_file(p)))
    rows.sort()
    lines = ["# d063 workspace manifest", "",
             f"workspace: {workspace}",
             f"generated: {time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())}", "",
             "| file | bytes | sha256 |", "| --- | --- | --- |"]
    for rel, sz, h in rows:
        lines.append(f"| `{rel}` | {sz} | `{h}` |")
    with open(os.path.join(workspace, "logs", "MANIFEST.md"), "w") as fh:
        fh.write("\n".join(lines) + "\n")
    return rows

def main():
    ap = argparse.ArgumentParser(description="d063 bounded six-agent cloud stress runner")
    ap.add_argument("--base-dir", default="/tmp", help="where to create the unique workspace")
    ap.add_argument("--workdir", default=None, help="explicit workspace path (must not exist)")
    ap.add_argument("--archive", default=None, help="use a local archive instead of downloading")
    ap.add_argument("--python", action="append", default=None,

    ap.add_argument("--download-timeout", type=int, default=DEFAULT_DOWNLOAD_TIMEOUT)
    ap.add_argument("--suite-timeout", type=int, default=DEFAULT_SUITE_TIMEOUT)
    ap.add_argument("--smoke-timeout", type=int, default=DEFAULT_SMOKE_TIMEOUT)

    ap.add_argument("--min-free-bytes", type=int, default=MIN_FREE_BYTES)

    deadline = overall_start + args.overall_deadline
    failures = []
    summary = {"commit": COMMIT, "archive_url": ARCHIVE_URL,
               "max_concurrent": MAX_CONCURRENT,
               "expected_suite_tests": EXPECTED_SUITE_TESTS}

    def remaining():
        return deadline - time.monotonic()

    def stage_timeout(cap):
        return max(1, int(min(cap, remaining())))

    # 1. workspace
    if args.workdir:
        ws = args.workdir
        if os.path.exists(ws):
            log(f"FATAL: workspace {ws} already exists; refusing to reuse")
            return 1
        os.makedirs(ws)
    else:
        os.makedirs(args.base_dir, exist_ok=True)

    os.makedirs(os.path.join(ws, "logs"), exist_ok=True)
    log(f"workspace: {ws}")
    summary["workspace"] = ws

    # 1b. verify the fixed payload identity before anything runs
    if not os.path.isfile(STRESS_CHECKS_PY):
        log(f"FAIL-CLOSED: payload {STRESS_CHECKS_PY} not found")
        return 1
    payload_sha = sha256_file(STRESS_CHECKS_PY)
    summary["payload_sha256"] = payload_sha
    if payload_sha != STRESS_CHECKS_SHA256:
        log(f"FAIL-CLOSED: payload sha256 {payload_sha} != pinned {STRESS_CHECKS_SHA256}")
        return 1
    log(f"payload stress_checks.py sha256 OK: {payload_sha}")

    # 2. env before + resource gate
    env_before = capture_env("before", ws)
    print_env(env_before)
    with open(os.path.join(ws, "logs", "env_before.json"), "w") as fh:
        json.dump(env_before, fh, indent=2)
    if env_before["workspace_fs_free_bytes"] is None:
        failures.append("cannot determine free space on workspace filesystem")
    elif env_before["workspace_fs_free_bytes"] < args.min_free_bytes:
        failures.append(f"insufficient free space: {env_before['workspace_fs_free_bytes']} "
                        f"< {args.min_free_bytes} bytes")
    if failures:
        log("FAIL-CLOSED (preflight): " + "; ".join(failures))
        return 1

    # 3. archive
    tgz = os.path.join(ws, "archive.tar.gz")
    try:
        if args.archive:
            size = copy_archive(args.archive, tgz, MAX_ARCHIVE_BYTES)
            log(f"copied archive {args.archive} ({size} bytes)")
        else:
            size = download_archive(ARCHIVE_URL, tgz, MAX_ARCHIVE_BYTES, args.download_timeout)
            log(f"downloaded archive ({size} bytes)")
        summary["archive_bytes"] = size
        summary["archive_sha256"] = sha256_file(tgz)
    except Exception as exc:  # noqa: BLE001
        log(f"FAIL-CLOSED (archive): {type(exc).__name__}: {exc}")
        return 1

    # 4. extract
    extract_dir = os.path.join(ws, "extract")
    try:
        os.makedirs(extract_dir, exist_ok=True)
        members = safe_extract(tgz, extract_dir)
        src = find_src_root(extract_dir)
        log(f"extracted {members} members; src root = {os.path.basename(src)}")
        summary["members"] = members
    except Exception as exc:  # noqa: BLE001
        log(f"FAIL-CLOSED (extract): {type(exc).__name__}: {exc}")
        return 1

    # 5. identity (BEFORE executing anything)
    problems, extra = verify_identities(src, EXPECTED_BLOBS)
    with open(os.path.join(ws, "logs", "identity.txt"), "w") as fh:
        fh.write(f"commit: {COMMIT}\nexpected executable files: {len(EXPECTED_BLOBS)}\n")
        fh.write("problems:\n" + ("\n".join(problems) if problems else "  (none)") + "\n")
        fh.write("extra files:\n" +
                 ("\n".join("  " + e for e in extra) if extra else "  (none)") + "\n")
    if extra:
        log(f"note: {len(extra)} file(s) present beyond the pinned set: {extra[:5]}")
    if problems:
        log("FAIL-CLOSED (identity):")
        for p in problems:
            log(f"  {p}")
        return 1
    log(f"identity OK: all {len(EXPECTED_BLOBS)} executable files match pinned blobs")

    # 6. interpreters + committed suite
    pythons = discover_pythons(args.python)
    summary["pythons"] = pythons
    log(f"interpreters found: {[p['name'] + '=' + p['version'] for p in pythons]}")
    if args.python:
        for want in args.python:
            if not any(p["name"] == want for p in pythons):
                log(f"note: requested interpreter {want!r} is unavailable")
    if not pythons:
        log("FAIL-CLOSED: no Python interpreter available")
        return 1

    env = build_env(ws, src)
    suite_results = {}
    for py in pythons:
        tag = py["name"]
        log(f"=== committed suite on {tag} ({py['version']}) ===")
        r = run([py["path"], "-m", "unittest", "discover", "-s", "tests", "-v"],
                cwd=src, env=env, timeout=stage_timeout(args.suite_timeout))
        parsed = parse_unittest(r["out"] + r["err"])
        r.update(parsed)
        with open(os.path.join(ws, "logs", f"suite_{tag}.out"), "w") as fh:
            fh.write(f"$ {' '.join(r['cmd'])}\n\n{r['out']}\n--- stderr ---\n{r['err']}\n")
            fh.write(f"EXIT={r['rc']} timed_out={r['timed_out']} seconds={r['seconds']}\n")
        suite_results[tag] = r
        log(f"  ran={parsed['ran']} skipped={parsed['skipped']} ok={parsed['ok']} "
            f"failed={parsed['failed']} exit={r['rc']} timeout={r['timed_out']}")

def _sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def _write(path: str, data: bytes):
    os.makedirs(os.path.dirname(path), exist_ok=True)

def _rss_kb():
    try:
        with open("/proc/self/status") as fh:
            for line in fh:
                if line.startswith("VmRSS:"):
                    return int(line.split()[1])
    except OSError:
        return None
    return None


def _nfd():
    try:
        return len(os.listdir("/proc/self/fd"))
    except OSError:
        return None

def build_roots(ws: str):
    """Create bounded, reused synthetic fixtures. Returns (roots, outside, bytes)."""
    roots = []
    total = 0
    outside = os.path.join(ws, "outside")
    os.makedirs(outside, exist_ok=True)
    _write(os.path.join(outside, "secret.txt"), b"SECRET-OUTSIDE-DATA")
    total += len(b"SECRET-OUTSIDE-DATA")

    for i in range(N_AGENTS):
        root = os.path.join(ws, "roots", f"r{i}")
        os.makedirs(os.path.join(root, "sub"), exist_ok=True)
        os.makedirs(os.path.join(root, "savepool"), exist_ok=True)
        fixtures = {
            "readme.txt": b"d063 fixture readme\n" * 4,
            "data.bin": bytes(range(256)) * 16,          # 4096 bytes
            "short.bin": b"abcde",                        # 5 bytes
            "empty.bin": b"",                             # 0 bytes
            "big.bin": b"B" * 262144,                     # 256 KiB
            "sub/note.txt": b"nested note\n",
        }
        for name, data in fixtures.items():

            total += len(data)

def build_policies(roots):
    from drive_tool.service_policy import AgentPolicy, SourceGrant
    policies = []
    for i, (token, alias, can_save, mrl, mwl) in enumerate(AGENT_SPECS):
        ops = {"find", "list", "read"} | ({"save"} if can_save else set())
        grant = SourceGrant(
            alias=alias,
            root=roots[i],
            operations=frozenset(ops),
            max_read_bytes=max(mrl, 1),
            max_write_bytes=max(mwl, 1),
        )
        policies.append(
            AgentPolicy(
                agent_id=f"agent{i}",
                token_sha256=hashlib.sha256(token.encode("ascii")).digest(),
                grants=(grant,),
            )
        )
    return tuple(policies)

def _do_call(svc, agent, rng, tag_counts):
    """Perform exactly one backend call for `agent`. Returns an outcome tag."""
    from drive_tool.file_service import ServiceError

    token, alias, can_save, mrl, mwl = AGENT_SPECS[agent]
    roll = rng.random()
    try:
        if roll < 0.05:
            # deliberate denial: use another agent's alias (alias isolation)
            other = (agent + 1) % N_AGENTS
            svc.list(token, AGENT_SPECS[other][1], "root", limit=5)
            return "unexpected_allow"
        if roll < 0.28:
            svc.list(token, alias, "root", limit=rng.choice([5, 20, 100]))
            return "list"
        if roll < 0.50:
            svc.find(token, alias, "root",
                     name_contains=rng.choice(["data", "readme", "sub", ".txt"]),
                     limit=rng.choice([5, 20]),
                     max_depth=rng.choice([0, 1, 4]))
            return "find"
        if roll < 0.85:
            fid = rng.choice(READ_IDS)
            length = rng.choice([1, 64, 1024, 4096, 65536, 262144])
            length = min(length, mrl)
            offset = rng.choice([0, 1, 100, 4096, SPARSE_SIZE - 65536])
            svc.read(token, alias, fid, offset=offset, length=length)
            return "read"
        # save into a bounded reused pool
        slot = rng.randrange(SAVE_POOL)
        size = rng.choice([0, 1, 7, 256, 1024])
        size = min(size, mwl)
        data = bytes([slot & 0xFF]) * size
        overwrite = rng.random() < 0.5
        svc.save(token, alias, f"p/savepool/f{slot:02d}.bin", data,
                 overwrite=overwrite)
        return "save"
    except ServiceError as exc:
        return "err:" + exc.code


def run_workload(calls: int, seed: int, deadline: float):
    from concurrent.futures import ThreadPoolExecutor
    ws = os.environ["D063_WORKSPACE"]
    roots, outside, fixture_bytes = build_roots(ws)
    policies = build_policies(roots)
    fd0, rss0, thr0 = _nfd(), _rss_kb(), threading.active_count()
    svc = make_service(policies)

    counts = {}
    lock = threading.Lock()
    unexpected = []
    peak = {"n": thr0}
    start = time.monotonic()

    def worker(w):
        rng = random.Random(seed * 1000 + w)
        local = {}
        for ticket in range(w, calls, N_AGENTS):
            if time.monotonic() > deadline:
                break
            try:
                tag = _do_call(svc, ticket % N_AGENTS, rng, local)
            except BaseException as exc:  # noqa: BLE001
                tag = "EXC:" + type(exc).__name__
                with lock:
                    unexpected.append(tag)
            local[tag] = local.get(tag, 0) + 1
            n = threading.active_count()
            if n > peak["n"]:
                with lock:
                    if n > peak["n"]:
                        peak["n"] = n
        with lock:
            for k, v in local.items():
                counts[k] = counts.get(k, 0) + v

    with ThreadPoolExecutor(max_workers=MAX_CONCURRENT) as ex:
        list(ex.map(worker, range(MAX_CONCURRENT)))

    thr_peak = peak["n"]
    closed = svc.close(timeout=10.0)
    fd1, rss1 = _nfd(), _rss_kb()

    completed = sum(v for k, v in counts.items())
    leaked = _scan_temp_leaks(roots)
    summary = {
        "calls_requested": calls,
        "calls_completed": completed,
        "outcomes": dict(sorted(counts.items())),
        "unexpected": unexpected,
        "closed_ok": bool(closed),
        "fd_before": fd0, "fd_after": fd1,
        "rss_before_kb": rss0, "rss_after_kb": rss1,
        "threads_before": thr0, "threads_peak": thr_peak,
        "temp_leaks": leaked,
        "fixture_bytes": fixture_bytes,
        "seconds": round(time.monotonic() - start, 2),
    }

    failures = []
    if unexpected:
        failures.append(f"{len(unexpected)} unexpected exceptions: {unexpected[:3]}")
    if completed != calls:
        failures.append(f"completed {completed} != requested {calls}")
    if counts.get("unexpected_allow"):
        failures.append("alias isolation leaked an allow")
    if not closed:
        failures.append("service.close() did not return True")
    if fd0 is not None and fd1 is not None and fd1 > fd0 + MAX_FD_SLACK:
        failures.append(f"fd growth {fd0}->{fd1} exceeds slack {MAX_FD_SLACK}")
    if rss0 is not None and rss1 is not None and (rss1 - rss0) > MAX_RSS_GROWTH_KB:
        failures.append(f"rss growth {rss1 - rss0} KiB exceeds {MAX_RSS_GROWTH_KB}")
    if thr_peak > thr0 + MAX_CONCURRENT + MAX_THREAD_SLACK:
        failures.append(f"thread peak {thr_peak} exceeds bound")
    if leaked:
        failures.append(f"leaked temp files: {leaked[:3]}")
    if fixture_bytes > MAX_FIXTURE_BYTES:
        failures.append(f"fixture bytes {fixture_bytes} exceeds {MAX_FIXTURE_BYTES}")
    summary["failures"] = failures
    return summary


def _scan_temp_leaks(roots):
    from drive_tool.local_write import TEMP_PREFIX
    leaks = []
    for root in roots:
        for dp, dns, fns in os.walk(root):
            for fn in fns + dns:
                if fn.startswith(TEMP_PREFIX):
                    leaks.append(os.path.join(dp, fn))
    return leaks

def _expect_error(fn, code, label, failures):

                          length=64).data

       f"published={published!r}")

    fd1 = _nfd()
    if fd0 is not None and fd1 is not None:

        "failures": failures,
        "outcomes_seen": seen["codes"],
        "seconds": round(time.monotonic() - t0, 2),
    }

# d063 — bounded six-agent cloud stress runner (GitHub-delivered)

Self-contained, **stdlib-only** runner for the director to execute on the
separate target instance `instance-20261015-210506` through the existing Cloud
Shell SSH. It validates the **pinned** accepted candidate

```
Dolesh-avkalan/drive-tool @ b7f8d17b9dcc4aec68d54b38f994290e05264e67
```

by downloading its GitHub archive, verifying identities **before** executing,
and running the committed 506 suite plus a bounded six-agent backend smoke,
a bounded six-agent backend soak (gated on the smoke) and raw edge checks.

It does **not** run on the target from the author's side. **No target result is
claimed or fabricated here.** This is *library* stress against real POSIX
fixtures — it is **not** an HTTP client/service acceptance test (there is no
HTTP adapter in this pinned revision; an HTTP smoke/soak will need a separately
pinned revision after the HTTP transport is accepted).

## The command

```
python3 run_cloud_stress.py
```

That is the whole invocation. It creates a fresh unique workspace under `/tmp`
(`/tmp/d063-b7f8d17b-<uniq>`), runs everything, prints `RESULT: PASS` or
`RESULT: FAIL (fail-closed)`, and exits `0` only on success. The workspace is
**kept** for retrieval; the exact packaging and cleanup commands are printed at
the end.

## Requirements / assumptions

- A POSIX host with Python 3 (the runner itself is stdlib-only; it then runs the
  suite on every interpreter it finds: `python3.14`, `3.13`, `3.12`, `3.11`,
  `3.10`, `3`). The target's Python 3.13.5 is picked up automatically.
- Outbound HTTPS to `codeload.github.com` for the pinned archive, **or** a
  pre-staged archive via `--archive <path>` (e.g. if the VM has no egress).
- Nothing is installed, cloned, mounted or written outside the workspace. No
  credentials, no cloud API, no reboot, no service change and no public binding.
  Real storage roots are never touched; fixtures are synthetic.

## What it checks, in order (each gate fails closed)

1. **Workspace + payload identity** — creates a new unique dir; verifies the
   fixed payload `stress_checks.py` SHA-256 before anything runs; requires free
   space (default ≥ 20 MiB).
2. **Environment (before)** — hostname, `uname`, OS release, interpreter, free
   space, memory, `nproc`.
3. **Pinned archive** — downloads the codeload tar.gz of the pinned commit with
   a 50 MiB byte cap and a wall-clock timeout (default 120 s).
4. **Guarded extraction** — rejects absolute paths, `..` components, symlink/
   hardlink members, non-regular members; caps member count (2000), per-file
   size (5 MiB) and total uncompressed size (50 MiB).
5. **Identity** — recomputes the git blob sha1 of all **26** pinned executable
   source/test files and compares to the commit's known hashes **before**
   executing anything. Any mismatch or missing file aborts with the offending
   path.
6. **Committed 506 suite** — `python -m unittest discover -s tests` on each
   available interpreter, under a wall-clock timeout. Records full
   stdout/stderr, exit, count and skips; fails closed if the count ≠ 506.
7. **Six-agent backend SMOKE** — 600 backend calls across 6 authenticated
   agents with real rooted adapters and grant policies.
8. **Six-agent backend SOAK** — 10000 backend calls, run **only if the smoke
   passes**.
9. **Raw edge checks** — grants, races, refusal, caps, EOF, leases and a
   clearly-labelled simulated post-publish fault.
10. **Environment (after)**, a **workspace-size gate** (≤ 128 MiB) and a
    **SHA-256/length manifest** of the whole workspace, plus `summary.json`.

## Fixed allow-listed actions only

The runner executes exactly one payload (`stress_checks.py`, identity-pinned)
with a fixed action token — `smoke`, `soak` or `edges` — plus the extracted
source dir. There is **no arbitrary shell/job payload**, no command string from
config, no credential and no cloud API call.

## Pass / fail / resource thresholds (defined before execution)

| threshold | value |
| --- | --- |
| concurrent operations | ≤ 6 |
| workspace size | ≤ 128 MiB |
| new fixture data | ≤ 32 MiB |
| archive / uncompressed / per-file | 50 / 50 / 5 MiB |
| archive members | ≤ 2000 |
| smoke / soak calls | exactly 600 / exactly 10000 completed |
| committed suite | exactly 506 tests, `OK`, exit 0, every interpreter |
| FD table after close | returns to baseline (+2 slack) |
| leaked temp files | none (`TEMP_PREFIX` scan of every root) |
| thread peak | ≤ baseline + 6 (+2 slack) |
| RSS growth (per stage) | ≤ 96 MiB |
| per-stage wall limits | download 120 s, suite 240 s, smoke 180 s, soak 420 s, edges 120 s |
| overall runner deadline | 900 s (< 20 min hard outer operator timeout) |

Recognized error codes counted exactly: `denied`, `exists`, `too_large`,
`invalid`, `not_found`, `unavailable`, `busy`, `timeout`. Any unexpected
exception fails the stage.

## Raw edge checks (stage `edges`)

alias isolation · readonly denial · path-traversal refusal · symlink-component
refusal (+ symlink omitted from listing) · grant-specific read/write caps ·
service read cap · exact-window / at-EOF / past-EOF / short / empty reads ·
large sparse range · empty-data save round-trip · concurrent no-overwrite with
exactly one winner and exact full payload · active-lease drain with idempotent
`close()` and post-close `unavailable` · **simulated** post-publish fault
injection (documented `LocalWriter._os_link` seam) asserting
`outcome_uncertain=True`, `retry_safe=False`, and that the file *was* published
before the error · FD baseline · no temp leaks.
